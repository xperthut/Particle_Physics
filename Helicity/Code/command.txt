# Helicity 2026 pipeline -- every command, in execution order.
# Run from Helicity/Code. Each step reads the CSVs written by the steps before it.
# Use the qsar env (the saved models were pickled with sklearn 1.8), e.g.
#   ~/miniconda3/envs/qsar/bin/python train.py ...


# 1. Model selection for M(omega, P) on the real data (All_p_w_m.csv)
#    -> model_selection_fits.csv / _summary.csv / _rmse_per_P.csv / _bootstrap.csv
#    then the best model's curves on omega = 0..20 -> M_curves_<best>.csv, where <best>
#    has the highest bootstrap QAICc win fraction (ties: lowest summed QAICc)
#    (--n_boot 0 skips the Exp bootstrap; <best> then = lowest summed QAICc)
python model_selection.py --n_boot 200
python model_selection.py --curves


# 2. Synthetic data: M at P=3 from M_curves_<best>.csv, M1..M5 and I from estimate_M_values,
#    then Exp replicas around them. Builds All_p_w_m.csv from rpitd/*.txt only if missing.
#    -> new_M.csv, I_omega_estimated.csv, loss_val_FM_{fm}.csv, M_real_vs_est_FM_{fm}.csv,
#       new_M_S.csv, generated_Synthetic_exp_data.csv
#    also real-data diagnostics: mean_median_and_qq.png, Mean_vs_W_FM_*.png, Std_vs_W_FM_*.png,
#    linearity_check.csv  (--diagnostics_only writes just these, no synthetic data)
python synthetic_data_generator.py


# 3. Sliding-window train frames, per FM
#    -> generated_Synthetic_exp_data_{fm}.csv, Synthetic_train_data_{with,without}_exp_{fm}.csv
python prepare_train_test_data.py


# 4. Train every model type, both variants
python train.py --fm all --with_exp_id 'No' --model_type 'GB'
python train.py --fm all --with_exp_id 'No' --model_type 'XGB'
python train.py --fm all --with_exp_id 'No' --model_type 'RF'
python train.py --fm all --with_exp_id 'No' --model_type 'NN'
python train.py --fm all --with_exp_id 'No' --model_type 'GP'

python train.py --fm all --with_exp_id 'Yes' --model_type 'GB'
python train.py --fm all --with_exp_id 'Yes' --model_type 'XGB'
python train.py --fm all --with_exp_id 'Yes' --model_type 'RF'
python train.py --fm all --with_exp_id 'Yes' --model_type 'NN'
python train.py --fm all --with_exp_id 'Yes' --model_type 'GP'

# Single ensemble, e.g. FM 9 only:
#   python train.py --fm 9 --with_exp_id 'Yes' --model_type 'XGB'
#
# Writes, per FM:
#   model/2026/{fm}/{model_type}_exp_{with_exp_id}.sav               (list of per-P fitted estimators)
#   model/2026/{fm}/{model_type}_exp_{with_exp_id}_best_params.json  (per-P best hyperparameters from BayesSearchCV)
#
# Speed knobs (defaults: n_iter=20, cv_splits=5, search_sample_size=20000):
# the search runs on at most --search_sample_size rows per P with --cv_splits
# forward-in-omega folds (train on w5 <= cut, validate on the next omega block,
# see OmegaBlockSplit in model.py); the winning hyperparameters are then refit on the FULL
# per-P data, so these only trade search precision for speed. GB is the slowest
# (sequential boosting) -- lower --n_iter/--search_sample_size for it first, e.g.:
#   python train.py --fm 9 --with_exp_id 'Yes' --model_type 'GB' --n_iter 10 --search_sample_size 10000
#
# NN (MLP, standardised inputs/target) uses the same BayesSearchCV knobs.
# GP ignores --n_iter/--cv_splits/--search_sample_size: its kernel is tuned by
# maximum likelihood on --gp_max_samples rows per P (default 2000; cost ~ n^3):
#   python train.py --fm 9 --with_exp_id 'No' --model_type 'GP' --gp_max_samples 3000
#
# Steps 5-7 use whichever of GB, RF, XGB, NN, GP have a saved .sav and skip the rest.


# 5. Training error (RMSE / R2 per FM, model, variant) -> printed
python train_error.py


# 6. Roll the models forward in omega from the synthetic seed
#    -> Real_data_projected_{fm}_no_exp_v2.csv / _with_exp_v2.csv
python reconstruct_real_data.py
python reconstruct_real_data.py --exp_id Yes
#    real-anchored replicas + GP latent draws, needed by continuum_M_extrapolation.py (step 9)
#    -> Real_data_projected_{fm}_no_exp_anchored_v2.csv, ..._anchored_draws_v2.csv
python reconstruct_real_data.py --anchor_real --n_draws 200 --models GP


# 7. M -> I(omega) per FM, test error vs real data, model ranking
#    -> Projected_M_mean_err_no_exp_{fm}.csv (Exp mean +- jackknife error),
#       Projected_I_no_exp_{fm}.csv, test_error_summary_no_exp.csv,
#       test_error_ranking_no_exp.csv, train_test_error_no_exp.csv,
#       best_model_I_no_exp_{model}.csv
python test_error.py


# 8. One model's I(omega) for all FM on one plot -> I_omega_all_FM_{model}_no_exp.csv
python combine_I_omega.py --model GP


# 9. Continuum limit a -> 0, all on the overall #1 model of train_test_error_no_exp.csv
#    pointwise on the best model's I (best_model_I_no_exp_<best>.csv) -> I_omega_continuum.csv
python continuum_extrapolation.py
#    jackknife over Exp on I -> I_omega_continuum_jk_<best>.csv
python continuum_I_jackknife.py --source roll
#    on M first, then I, jackknife stat + fit/model/form errors (anchored needs the
#    --anchor_real run of step 6) -> I_omega_continuum_M_<best>_anchored.csv
python continuum_M_extrapolation.py --source anchored --n_draws 500
